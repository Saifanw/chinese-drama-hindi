rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /comments/{commentId} {
      // Anyone can read comments and submit a validated comment.
      allow read: if true;
      allow create: if
        request.resource.data.keys().hasAll(['videoId','name','text','createdAt']) &&
        request.resource.data.keys().hasOnly(['videoId','name','text','createdAt']) &&
        request.resource.data.videoId is string &&
        request.resource.data.videoId.size() > 0 &&
        request.resource.data.videoId.size() <= 100 &&
        request.resource.data.name is string &&
        request.resource.data.name.size() > 0 &&
        request.resource.data.name.size() <= 50 &&
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() <= 1000 &&
        request.resource.data.createdAt == request.time;
      // Visitors cannot edit or delete comments after posting.
      allow update, delete: if false;
    }
  }
}
